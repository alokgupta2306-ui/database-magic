CREATE TYPE Person AS ( first_name TEXT, last_name TEXT, age INT );
CREATE OR REPLACE FUNCTION get_full_name(p Person) RETURNS TEXT AS $$ BEGIN RETURN p.first_name || ' ' || p.last_name; END; $$ LANGUAGE plpgsql;
CREATE TABLE students ( id SERIAL PRIMARY KEY, info Person );
